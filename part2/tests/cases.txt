# Тестовые наборы для run_tests.sh
# Формат строки:  ожидание | название | аргументы
#   ожидание: done  - завершение после обработки всех авто
#             limit - завершение по лимиту времени
#             error - ошибка в параметрах (код выхода 1, журнал не проверяется)
# Ко всем аргументам спереди добавляется "-d 1 -t 0 -L <журнал>": такт 1 мс и без лимита времени
# (аргументы набора идут после и могут их переопределить).

# --- Базовые сценарии ---
done  | параметры по умолчанию            |
done  | без очереди                       | -q 0
done  | большая очередь, долгое ожидание  | -q 10 -w 100
done  | нулевое время ожидания            | -a 0 -b 0 -p 1 -P 1 -w 0
done  | нулевое время стоянки             | -p 0 -P 0
done  | нет автомобилей                   | -c 0 -C 0 -S 0
done  | нет мест для двух типов           | -r 1 -l 0 -s 0 -c 10 -C 5 -S 5 -q 5 -w 3
done  | нет мест совсем                   | -r 0 -l 0 -s 0 -q 3 -w 2
done  | пример из readme                  | -q 5 -r 1 -l 1 -s 0 -p 15 -P 30 -c 20

# --- Стратегии (одинаковая нагрузка) ---
done  | стратегия fifo                    | -c 50 -C 50 -S 50 -a 0 -b 1 -q 20 -w 15 -g fifo
done  | стратегия lifo                    | -c 50 -C 50 -S 50 -a 0 -b 1 -q 20 -w 15 -g lifo
done  | стратегия random                  | -c 50 -C 50 -S 50 -a 0 -b 1 -q 20 -w 15 -g random
done  | стратегия short                   | -c 50 -C 50 -S 50 -a 0 -b 1 -q 20 -w 15 -g short

# --- Нагрузочные ---
done  | 500 авто, 200 мест                | -r 200 -q 100 -c 300 -C 100 -S 100 -a 0 -b 0
done  | гонка «таймаут и назначение»      | -r 2 -l 2 -s 2 -q 30 -w 2 -c 60 -C 60 -S 60 -a 0 -b 0 -p 1 -P 3
done  | все прибывают одновременно        | -a 0 -b 0 -q 50 -w 30

# --- Лимит времени ---
limit | лимит 30 тактов, долгая стоянка   | -t 30 -p 50 -P 60
limit | лимит при полной очереди          | -t 20 -q 10 -w 100 -a 0 -b 0 -p 30 -P 40
done  | без лимита, долгая стоянка        | -t 0 -p 20 -P 30

# --- Ошибки в параметрах ---
error | min > max (прибытие)              | -a 3 -b 2
error | min > max (стоянка)               | -p 5 -P 1
error | отрицательный лимит               | -t -5
error | отрицательная очередь             | -q -1
error | такт 0 мс                         | -d 0
error | неизвестная стратегия             | -g bogus
error | неизвестный флаг                  | -x
